import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, mean_squared_error
import os

# Загрузим обработанные дневные данные из папки
data_dir = 'data'

# Быстрая сборка дневных PM2.5
pm25_path = os.path.join(data_dir, 'pm2_5.csv')
df = pd.read_csv(pm25_path)
df = df[df['city'].str.contains('Pavlodar', case=False, na=False)]
df['datetime_utc'] = pd.to_datetime(df['datetime_utc'])
df_daily = df.set_index('datetime_utc').resample('D')['value_ugm3'].mean().to_frame(name='pm2_5')
df_daily = df_daily.dropna()

# Подготовка признаков (Features): используем лаги (предыдущие дни)
df_daily['pm2_5_lag1'] = df_daily['pm2_5'].shift(1)
df_daily['pm2_5_lag2'] = df_daily['pm2_5'].shift(2)
df_daily = df_daily.dropna()

X = df_daily[['pm2_5_lag1', 'pm2_5_lag2']]
y = df_daily['pm2_5']

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, shuffle=False)
print(f"Тренировочный набор: {len(X_train)} дней, Тестовый: {len(X_test)} дней")

# Обучение модели
model = LinearRegression()
model.fit(X_train, y_train)
y_pred = model.predict(X_test)

mae = mean_absolute_error(y_test, y_pred)
print(f"Mean Absolute Error (MAE): {mae:.2f}")

# Визуализация предсказаний
plt.figure(figsize=(12, 5))
plt.plot(y_test.index, y_test, label='Фактические значения', color='blue')
plt.plot(y_test.index, y_pred, label='Предсказание модели', color='red', linestyle='--')
plt.title('Прогноз PM2.5 (Линейная регрессия с лагами)')
plt.legend()
plt.show()
